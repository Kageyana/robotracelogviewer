        % ログデータを取得する
        function isSuccess = getLogdata(app)
            isSuccess = false; % 全処理が成功するまで未完了とする

			try
				% ログファイル読み込みと最低限の行数チェックを例外処理で保護
                if isempty(app.LogDropDown.ItemsData)
                    error("logAnalysis:NoLogFiles","読み込み可能なログファイルがありません。");
                end
                filePath = fullfile(app.rootDir,app.LogDropDown.Value);
                % 先頭2行だけを読み、メタ情報は最初から文字列として扱う
                [parameterData,columnNames] = readLogHeaders(app,filePath);
                importOptions = detectImportOptions(filePath, ...
                    'VariableNamesLine',2,'VariableNamingRule','preserve');
                importOptions.DataLines = [3 Inf];
                if ~all(ismember(columnNames,string(importOptions.VariableNames)))
                    error("logAnalysis:InvalidHeader","列名行と読込オプションの列名が一致しません。");
                end
                % 末尾カンマによる空列を本体テーブルへ取り込まない
                importOptions.SelectedVariableNames = cellstr(columnNames);
                logData = readtable(filePath,importOptions);
                if height(logData) < 2
                    error("logAnalysis:InsufficientLogRows","ログファイルの行数が2行未満です。");
                end

                % 時間[s]は有限・非負で、各ログ行で必ず増加すること
                timeData = evaluateLogEquation(app,logData,app.timeEditField.Value,"時間");
                intervals = diff(timeData);
                if any(timeData < 0) || any(~isfinite(intervals)) || any(intervals <= 0)
                    error("logAnalysis:InvalidTime", ...
                        "時間は非負かつ単調増加が必要です。重複・逆行・不正な時間間隔を確認してください。");
                end
                dtData = [intervals;intervals(end)];

                % 速度[mm/s]。diff時は換算後の値を時間間隔で微分する
                velocityData = evaluateLogEquation(app,logData,app.velocityEditField.Value,"速度");
                if app.diffCheckBox.Value
                    delta = diff(velocityData);
                    velocityData = [delta;delta(end)] ./ dtData;
                    if any(~isfinite(velocityData))
                        error("logAnalysis:InvalidVelocity","速度の差分計算結果にInfまたはNaNがあります。");
                    end
                end

                % 角速度は単位チェックボックスで指定したdeg/sまたはrad/s
                angularVelocityData = evaluateLogEquation(app,logData,app.angularVelocityEditField.Value,"角速度");
                if isempty(parameterData)
                    app.TextArea_2.Value = "";
                else
                    app.TextArea_2.Value = parameterData;
                end
                % 全ての読み込み・換算が成功した後にまとめて反映する
                app.log = logData;
                app.time = timeData;
                app.dt = dtData;
                app.velocity = velocityData;
                app.angularVelocity = angularVelocityData;
                app.parameter = parameterData;
                isSuccess = true;

			catch ME
				app.Lamp.Color = "red";	% 例外発生をランプで通知
				setTexttarea(app, {ME.identifier, ME.message});	% エラー内容をテキストエリアに表示
				isSuccess = false;
			end

			% 例外時は後続処理をスキップ
			if ~isSuccess
				return;
			end

			setTexttarea(app, {filePath, 'ログファイル読み込み完了'});
        end
        

        % UTF-8ログの先頭2行を読み、メタ情報と列名を文字列へ統一する
        % 引数 filePath: ログCSVのパス
        % 戻り値 parameters: 表示用の文字列列ベクトル、columnNames: 列名
        function [parameters,columnNames] = readLogHeaders(app,filePath)
            [fileId,message] = fopen(filePath,'r','n','UTF-8');
            if fileId < 0
                error("logAnalysis:HeaderReadFailed","ヘッダを読み込めません: %s",message);
            end
            cleanupFile = onCleanup(@()fclose(fileId));
            metadataLine = fgetl(fileId);
            columnsLine = fgetl(fileId);
            if ~ischar(metadataLine) || ~ischar(columnsLine)
                error("logAnalysis:InvalidHeader","ログにはメタ情報行と列名行が必要です。");
            end
            if ~isempty(metadataLine) && metadataLine(1)==char(65279)
                metadataLine = metadataLine(2:end); % UTF-8 BOMを除く
            end
            % %qで引用符内のカンマも扱い、数値・NaNを数値へ変換しない
            parameters = strings(0,1);
            if ~isempty(metadataLine)
                metadata = textscan(metadataLine,'%q','Delimiter',',','Whitespace','');
                parameters = string(metadata{1});
                parameters(ismissing(parameters)) = "";
                parameters = parameters(strlength(parameters)>0);
                parameters = parameters(:);
            end

            if isempty(columnsLine)
                error("logAnalysis:InvalidHeader","列名行が空です。");
            end
            columns = textscan(columnsLine,'%q','Delimiter',',','Whitespace','');
            columnNames = strtrim(string(columns{1}));
            while ~isempty(columnNames) && strlength(columnNames(end))==0
                columnNames(end) = []; % ログの末尾カンマを許容する
            end
            if isempty(columnNames) || any(ismissing(columnNames)) || ...
                    any(strlength(columnNames)==0) || numel(unique(columnNames))~=numel(columnNames)
                error("logAnalysis:InvalidHeader","列名が空、または重複しています。");
            end
            columnNames = columnNames(:);
        end
        % ログ列と定数の乗除算からなる換算式を検証・計算する
        % 引数 logData: 読込ログ、equation: 列名に*または/と数値を続ける式
        %      label: エラー表示用の項目名
        % 戻り値 data: 有限な実数の列ベクトル（double）
        function data = evaluateLogEquation(app,logData,equation,label)
            tokens = validateLogEquation(app,equation,label);
            column = tokens{1};
            if ~ismember(column,logData.Properties.VariableNames)
                error("logAnalysis:MissingColumn","%s換算式の列がありません: %s",label,column);
            end
            data = logData.(column);
            if ~isnumeric(data) || ~isreal(data) || ~isvector(data) || ...
                    numel(data) ~= height(logData) || any(~isfinite(data(:)))
                error("logAnalysis:InvalidColumn", ...
                    "%s換算式の列は、各行に有限な実数が必要です: %s",label,column);
            end
            data = double(data(:));
            for tokenIndex = 2:2:numel(tokens)
                operator = tokens{tokenIndex};
                operandText = tokens{tokenIndex+1};
                numericPattern = '^[+-]?(\d+(\.\d*)?|\.\d+)([eE][+-]?\d+)?$';
                operand = str2double(operandText);
                if isempty(regexp(operandText,numericPattern,'once')) || ...
                        ~isreal(operand) || ~isfinite(operand)
                    error("logAnalysis:InvalidEquation", ...
                        "%s換算式には有限な実数を指定してください: %s",label,operandText);
                end
                if strcmp(operator,'*')
                    data = data .* operand;
                elseif strcmp(operator,'/')
                    if operand == 0
                        error("logAnalysis:InvalidEquation","%s換算式でゼロ除算はできません。",label);
                    end
                    data = data ./ operand;
                else
                    error("logAnalysis:InvalidEquation","%s換算式の演算子が不正です: %s",label,operator);
                end
                if any(~isfinite(data))
                    error("logAnalysis:InvalidEquation","%s換算結果にInfまたはNaNがあります。",label);
                end
            end
        end
        % 換算式の構文と定数を検証する（ログ読込前の設定検証にも使用）
        % 引数 equation: 換算式、label: エラー表示用の項目名
        % 戻り値 tokens: 列名と演算子・定数のトークン
        function tokens = validateLogEquation(app,equation,label)
            equation = char(string(equation));
            tokens = regexp(strtrim(strrep(strrep(equation,'*',' * '),'/',' / ')), ...
                '\S+','match');
            if isempty(tokens) || mod(numel(tokens),2) ~= 1 || ...
                    isempty(regexp(tokens{1},'^[A-Za-z]\w*$','once'))
                error("logAnalysis:InvalidEquation", ...
                    "%s換算式は「列名 * 数値 / 数値」の形式で指定してください。",label);
            end
            for tokenIndex = 2:2:numel(tokens)
                operator = tokens{tokenIndex};
                operandText = tokens{tokenIndex+1};
                operand = str2double(operandText);
                numericPattern = '^[+-]?(\d+(\.\d*)?|\.\d+)([eE][+-]?\d+)?$';
                if ~ismember(operator,{'*','/'}) || ...
                        isempty(regexp(operandText,numericPattern,'once')) || ...
                        ~isreal(operand) || ~isfinite(operand) || ...
                        (strcmp(operator,'/') && operand == 0)
                    error("logAnalysis:InvalidEquation", ...
                        "%s換算式の演算子または定数が不正です: %s %s",label,operator,operandText);
                end
            end
        end

        % 設定CSVを全項目検証し、検証成功後にUIへ反映する
        % 引数 filePath: 設定CSVのパス
        % 戻り値 isSuccess: 読込・反映成功、settingsMap: 検証済み設定
        function [isSuccess,settingsMap] = loadSettings(app,filePath)
            isSuccess = false;
            settingsMap = [];
            try
                settingsTable = readtable(filePath,'TextType','string');
                if ~all(ismember({'key','value'},settingsTable.Properties.VariableNames))
                    error("logAnalysis:InvalidSettings","設定CSVにはkey,value列が必要です。");
                end
                keys = strtrim(string(settingsTable.key));
                values = strtrim(string(settingsTable.value));
                if any(ismissing(keys)) || any(ismissing(values)) || ...
                        any(strlength(keys)==0) || any(strlength(values)==0) || ...
                        numel(unique(keys)) ~= numel(keys)
                    error("logAnalysis:InvalidSettings","設定CSVに空の項目または重複キーがあります。");
                end
                candidate = containers.Map(cellstr(keys),cellstr(values));
                required = {'logFolderDir','timeEquation','velocityEquation','angularVelocityEquation', ...
                    'radsCheckBox','switchDirection','data1','data2','data3','analysisLeft','analysisRight', ...
                    'smoothLeft','smoothRight','windowLeft','windowRight','normalizeLeft','normalizeRight', ...
                    'detrendLeft','detrendRight','ylimMatch','plotSwitch','windowSlider'};
                missingKeys = required(~isKey(candidate,required));
                if ~isempty(missingKeys)
                    error("logAnalysis:InvalidSettings","設定キーが不足しています: %s",strjoin(missingKeys,', '));
                end
                % 旧3グラフ設定は新規2項目の既定値を補って読み込む
                extraKeys = {'data4','data5'};
                extraDefaults = {'gyroVal_Z','encCurrentCorr_p'};
                for itemIndex = 1:2
                    if ~isKey(candidate,extraKeys{itemIndex})
                        candidate(extraKeys{itemIndex}) = extraDefaults{itemIndex};
                    end
                end
                if ~isfolder(candidate('logFolderDir'))
                    error("logAnalysis:InvalidSettings","ログフォルダが存在しません: %s",candidate('logFolderDir'));
                end
                validateLogEquation(app,candidate('timeEquation'),"時間");
                validateLogEquation(app,candidate('velocityEquation'),"速度");
                validateLogEquation(app,candidate('angularVelocityEquation'),"角速度");
                if ~ismember(lower(candidate('radsCheckBox')),{'true','false'})
                    error("logAnalysis:InvalidSettings","radsCheckBoxはtrueまたはfalseで指定してください。");
                end
                optionKeys = {'switchDirection','smoothLeft','smoothRight','normalizeLeft', ...
                    'normalizeRight','detrendLeft','detrendRight','ylimMatch','plotSwitch'};
                optionControls = {app.SwitchDirection,app.smoothLDropDown,app.smoothRDropDown, ...
                    app.normalizeLSwitch,app.normalizeRSwitch,app.detrendLSwitch,app.detrendRSwitch, ...
                    app.ylimmatchSwitch,app.ShortCutplotSwitch};
                for itemIndex = 1:numel(optionKeys)
                    key = optionKeys{itemIndex};
                    if ~ismember(candidate(key),optionControls{itemIndex}.Items)
                        error("logAnalysis:InvalidSettings","%sの設定値が不正です: %s",key,candidate(key));
                    end
                end
                numberKeys = {'windowLeft','windowRight','windowSlider'};
                numberControls = {app.windowLSpinner,app.windowRSpinner,app.ShortCutwindowSlider};
                numberValues = zeros(1,3);
                for itemIndex = 1:3
                    key = numberKeys{itemIndex};
                    value = str2double(candidate(key));
                    limits = numberControls{itemIndex}.Limits;
                    if ~isreal(value) || ~isfinite(value) || value ~= fix(value) || ...
                            value < limits(1) || value > limits(2)
                        error("logAnalysis:InvalidSettings","%sは範囲内の整数で指定してください。",key);
                    end
                    numberValues(itemIndex) = value;
                end

                % 以下はすべて検証済み。途中までの設定反映を避ける
                app.timeEditField.Value = candidate('timeEquation');
                app.velocityEditField.Value = candidate('velocityEquation');
                app.angularVelocityEditField.Value = candidate('angularVelocityEquation');
                app.radsCheckBox.Value = strcmpi(candidate('radsCheckBox'),'true');
                for itemIndex = 1:numel(optionKeys)
                    optionControls{itemIndex}.Value = candidate(optionKeys{itemIndex});
                end
                for itemIndex = 1:3
                    numberControls{itemIndex}.Value = numberValues(itemIndex);
                end
                app.rootDir = string(candidate('logFolderDir'));
                [~,folderName] = fileparts(char(app.rootDir));
                app.FolderButton.Text = folderName;
                app.csvFilePathEditer.Value = fileparts(char(filePath));
                settingsMap = candidate;
                isSuccess = true;
            catch ME
                app.Lamp.Color = "red";
                setTexttarea(app,{"設定読込失敗",ME.identifier,ME.message});
            end
        end

        % ログ列の一覧に存在する保存済み選択だけを復元する
        % 引数 settingsMap: 検証済み設定
        function restoreLogSelections(app,settingsMap)
            keys = {'data1','data2','data3','data4','data5','analysisLeft','analysisRight'};
            controls = {app.Data1DropDown,app.Data2DropDown,app.Data3DropDown, ...
                app.Data4DropDown,app.Data5DropDown,app.DataAnalysisLDropDown,app.DataAnalysisRDropDown};
            for itemIndex = 1:numel(keys)
                key = keys{itemIndex};
                value = settingsMap(key);

                    controls{itemIndex}.Value = value;
                else
                    setTexttarea(app,{sprintf("設定項目%sの列%sがログにないため、現在の選択を使用します。",key,value)});
                end
            end
        end

        % ログファイル一覧の取得
        function isSuccess = getLogfiles(app)
            isSuccess = false;
            selectedFile = string(app.LogDropDown.Value);
            try
                if ~isfolder(app.rootDir)
                    error("MATLAB:cd:NonExistentFolder","The folder does not exist.");
                end
                fileList = struct2table(dir(fullfile(app.rootDir,'*.csv')));
                if isempty(fileList)
                    error("logAnalysis:NoLogFiles","読み込み可能なログファイルがありません。");
                end
                % ソート用の番号と読み込み用の元ファイル名を分離する
                fileNames = string(fileList.name);
                logNumbers = str2double(erase(fileNames,lettersPattern | characterListPattern(".")));
                valid = ~isnan(logNumbers) & fileList.bytes ~= 0;
                fileList = fileList(valid,:);
                fileNames = fileNames(valid);
                logNumbers = logNumbers(valid);
                if isempty(fileNames)
                    error("logAnalysis:NoLogFiles","読み込み可能なログファイルがありません。");
                end
                [~,order] = sort(logNumbers,'descend');
                fileList = fileList(order,:);
                fileNames = fileNames(order);
                logNumbers = logNumbers(order);
                fileSize = string(round(fileList.bytes/1000));
                fileDate = string(datetime(fileList.datenum,"ConvertFrom","datenum"),"yyyy/MM/dd");

                % 表示文字列が変わってもファイル名で選択を保持する
                app.LogDropDown.ItemsData = {};
                app.LogDropDown.Items = append(string(logNumbers),"    ",fileSize,"KB","    ",fileDate);
                app.LogDropDown.ItemsData = cellstr(fileNames);
                if ismember(selectedFile,fileNames)
                    app.LogDropDown.Value = char(selectedFile);
                else
                    % 選択ログが削除された場合は一覧の先頭を選ぶ
                    app.LogDropDown.Value = char(fileNames(1));
                end
                app.LogDropDown.Enable = "on";
                app.Lamp.Color = "green";
                app.TextArea.Value = "";
                isSuccess = true;
            catch ME
                app.LogDropDown.ItemsData = {};
                app.LogDropDown.Items = {'not found files'};
                app.LogDropDown.Enable = "off";
                app.Lamp.Color = "red";
                setTexttarea(app,{ME.identifier,ME.message});
                if strcmp(ME.identifier,'MATLAB:cd:NonExistentFolder')
                    app.FolderButton.Text = "not found folder";
                end
            end
        end

        % フォルダ変更を読み込みまで実行し、失敗したら以前の表示状態へ戻す
        % 引数 folder: 選択されたフォルダ（キャンセル時は0）
        % 戻り値 isSuccess: 更新が完了したか
        function isSuccess = changeLogFolder(app,folder)
            isSuccess = false;
            if isequal(folder,0) || isempty(folder)
                return;
            end
            propertiesToSave = {'rootDir','log','time','dt','velocity','angularVelocity', ...
                'parameter','course','shortCutCourse'};
            previous = cell(size(propertiesToSave));
            for itemIndex = 1:numel(propertiesToSave)
                previous{itemIndex} = app.(propertiesToSave{itemIndex});
            end
            controls = {app.LogDropDown,app.Data1DropDown,app.Data2DropDown,app.Data3DropDown, ...
                app.Data4DropDown,app.Data5DropDown,app.DataAnalysisLDropDown,app.DataAnalysisRDropDown, ...
                app.Plotflag1DropDown,app.Plotflag2DropDown,app.Plotflag3DropDown};
            items = cellfun(@(control)control.Items,controls,'UniformOutput',false);
            values = cellfun(@(control)control.Value,controls,'UniformOutput',false);
            itemsData = app.LogDropDown.ItemsData;
            logEnable = app.LogDropDown.Enable;
            folderLabel = app.FolderButton.Text;
            sliderLimits = app.TimesSlider.Limits;
            sliderValue = app.TimesSlider.Value;
            sliderTicks = app.TimesSlider.MajorTicks;
            sliderLabels = app.TimesSlider.MajorTickLabels;
            metadata = app.TextArea_2.Value;
            app.rootDir = string(folder);
            if reload(app,"ButtonPushed")
                [~,folderName] = fileparts(char(folder));
                app.FolderButton.Text = folderName;
                isSuccess = true;
                return;
            end
            for itemIndex = 1:numel(propertiesToSave)
                app.(propertiesToSave{itemIndex}) = previous{itemIndex};
            end
            app.LogDropDown.ItemsData = {};
            for itemIndex = 1:numel(controls)
                controls{itemIndex}.Items = items{itemIndex};
                if itemIndex ~= 1
                    controls{itemIndex}.Value = values{itemIndex};
                end
            end
            app.LogDropDown.ItemsData = itemsData;
            app.LogDropDown.Value = values{1};
            app.LogDropDown.Enable = logEnable;
            app.FolderButton.Text = folderLabel;
            app.TextArea_2.Value = metadata;
            app.TimesSlider.Limits = sliderLimits;
            app.TimesSlider.Value = sliderValue;
            app.TimesSlider.MajorTicks = sliderTicks;
            app.TimesSlider.MajorTickLabels = sliderLabels;
            % 読込後の描画失敗で消えた表示も、保存したデータから復元する
            if ~isempty(app.course)
                plotCourse(app);
                plotVariableGraphs(app);
                procData(app,"left");
            end
            app.Lamp.Color = "red";
        end

                bounds = [min(app.course.x)-padding max(app.course.x)+padding; ...
                    min(app.course.y)-padding max(app.course.y)+padding];

            app.UIAxisCourse.XLim = bounds(1,:);
            app.UIAxisCourse.YLim = bounds(2,:);
        end

        % コース座標の計算
        function isSuccess = calcCourseplot(app)
            isSuccess = false;
            if isempty(app.time) || isempty(app.velocity) || isempty(app.angularVelocity)
                return;
            end
            try
                % 換算式の出力単位に合わせて角度を積算する
                angle = cumtrapz(app.time,app.angularVelocity .* app.kgyro);
                if app.radsCheckBox.Value
                    angle = angle + deg2rad(app.rotate);
                    x = cumtrapz(app.time,app.velocity .* sin(angle));
                    y = cumtrapz(app.time,app.velocity .* cos(angle));
                else
                    angle = angle + app.rotate;
                    x = cumtrapz(app.time,app.velocity .* sind(angle));
                    y = cumtrapz(app.time,app.velocity .* cosd(angle));
                end
                if any(~isfinite(angle)) || any(~isfinite(x)) || any(~isfinite(y))
                    error("logAnalysis:InvalidCourse","軌跡計算結果にInfまたはNaNがあります。換算式と補正係数を確認してください。");
                end
                % timetable変数作成
                app.course = timetable(seconds(app.time),x,y,app.velocity,'VariableNames',{'x','y','velocity'});
                app.shortCutCourse = []; % 基準軌跡が変わったら旧座標を無効にする
                isSuccess = true;
            catch ME
                app.Lamp.Color = "red";     % ステータスランプ赤
                % エラー表示
                for num = 1:1:length(ME.stack)
                    setTexttarea(app, {ME.stack(num,1).name,num2str(ME.stack(num,1).line)})
                end
                setTexttarea(app, {ME.identifier,ME.message})
            end
        end

        % ショートカット軌跡座標の計算
        function isSuccess = calcShortCutplot(app)
            isSuccess = false;
            app.shortCutCourse = [];
            if isempty(app.course)
                return;
            end
            try
                % windowが1未満の場合は処理を中断
                app.ShortCutwindowSlider.Value = round(app.ShortCutwindowSlider.Value);
                if app.ShortCutwindowSlider.Value < 1
                    app.Lamp.Color = "red";     % ステータスランプ赤
                    setTexttarea(app, {"windowの値が1未満です", "1以上に設定してください"});
                    return;
                end
                % 座標算出
                x = movmean(app.course.x,app.ShortCutwindowSlider.Value);
                y = movmean(app.course.y,app.ShortCutwindowSlider.Value);
                % timetable変数作成
                app.shortCutCourse = timetable(app.course.Time,x,y,app.course.velocity,'VariableNames',{'x','y','velocity'});
                isSuccess = true;
            catch ME
                app.Lamp.Color = "red";     % ステータスランプ赤
                % エラー表示
                for num = 1:1:length(ME.stack)
                    setTexttarea(app, {ME.stack(num,1).name,num2str(ME.stack(num,1).line)})
                end
                setTexttarea(app, {ME.identifier,ME.message})
            end
        end
       
        % コース描画
        function isSuccess = plotCourse(app)
            isSuccess = false;
            if isempty(app.course)
                return;
            end
            % 表示するたびに現在の基準軌跡とwindowから座標を更新する
            if strcmp(app.ShortCutplotSwitch.Value,"On") && ~calcShortCutplot(app)
                return;
            end
            try
                % 初回だけ作成し、以後は座標と速度色を更新する
                previousNextPlot = app.UIAxisCourse.NextPlot;
                app.UIAxisCourse.NextPlot = 'add';
                cleanupAxes = onCleanup(@()set(app.UIAxisCourse,'NextPlot',previousNextPlot));
                if isempty(app.coursePlot) || ~isgraphics(app.coursePlot)
                    app.coursePlot = scatter(app.UIAxisCourse,app.course.x,app.course.y, ...
                        36,app.course.velocity, ...
                        'ButtonDownFcn',@(src,evnt)getTimeIndex(app,src,evnt));
                else
                    set(app.coursePlot,'XData',app.course.x,'YData',app.course.y, ...
                        'CData',app.course.velocity);
                end
                applyCourseLimits(app);
                daspect(app.UIAxisCourse,[1 1 1]);
                app.UIAxisCourse.CLimMode = 'auto';

                if isempty(app.courseColorbar) || ~isgraphics(app.courseColorbar)
                    app.courseColorbar = colorbar(app.UIAxisCourse);
                    app.courseColorbar.Label.String = 'velocity[mm/s]';
                    app.courseColorbar.Label.FontSize = 12;
                end

                if strcmp(app.ShortCutplotSwitch.Value,"On")
                    if isempty(app.shortCutPlot) || ~isgraphics(app.shortCutPlot)
                        app.shortCutPlot = scatter(app.UIAxisCourse,app.shortCutCourse.x, ...
                            app.shortCutCourse.y,'HitTest','off','PickableParts','none');
                    else
                        set(app.shortCutPlot,'XData',app.shortCutCourse.x, ...
                            'YData',app.shortCutCourse.y,'Visible','on');
                    end
                elseif ~isempty(app.shortCutPlot) && isgraphics(app.shortCutPlot)
                    app.shortCutPlot.Visible = 'off';
                end

                if ~plotFlagPoint(app)
                    return;
                end
                selectCourseIndex(app,app.TimesSlider.Value);
                isSuccess = true;
            catch ME
                app.Lamp.Color = "red";     % ステータスランプ赤
                % エラー表示
                for num = 1:1:length(ME.stack)
                    % setTexttarea(app, {ME.stack(num,1).name,num2str(ME.stack(num,1).line)})
                end
                setTexttarea(app, {ME.identifier,ME.message});
            end
        end

        % ログの取得、再プロット
        function isSuccess = reload(app,EventName)
            isSuccess = false;
            try
            if isempty(app.rootDir) || isequal(app.rootDir,0)
                folder = uigetdir;
                if isequal(folder,0)
                    return;
                end
                app.rootDir = folder;
            end
            if ~isempty(app.rootDir) && ~isequal(app.rootDir,0)
                if strcmp(EventName,"ButtonPushed")
                    if ~getLogfiles(app)
                        return;
                    end
                end
                if ~getLogdata(app)
                    return;
                end
                isSuccess = initializeLogView(app,[]);
            end
            catch ME
                app.Lamp.Color = "red";
                setTexttarea(app,{"ログ更新失敗",ME.identifier,ME.message});
            end
        end

                    axesList{itemIndex}.XLim = [0 app.time(end)];

            end
        end

        % データ処理
        function isSuccess = procData(app,pos)
            isSuccess = false;
            try
            % 両側を現在の設定で処理し、描画データと範囲計算を一致させる
            [varNameL,varDataL] = getAnalysisData(app,"left");
            [varNameR,varDataR] = getAnalysisData(app,"right");
            sides = {"left","right"};
            names = {varNameL,varNameR};
            data = {varDataL,varDataR};
            for sideIndex = 1:2
                yyaxis(app.UIAxesAnalysis,sides{sideIndex});
                ylabel(app.UIAxesAnalysis,names{sideIndex});
                if strcmp(names{sideIndex},"none")
                    if ~isempty(app.analysisPlots{sideIndex}) && isgraphics(app.analysisPlots{sideIndex})
                        app.analysisPlots{sideIndex}.Visible = 'off';
                    end
                else
                    app.analysisPlots{sideIndex} = updateDataLine(app,app.UIAxesAnalysis, ...
                        app.analysisPlots{sideIndex},data{sideIndex});
                end
                app.UIAxesAnalysis.YLimMode = "auto";
            end

            if strcmp(app.ylimmatchSwitch.Value,"On") && ...
                    ~strcmp(varNameL,"none") && ~strcmp(varNameR,"none")
                values = [varDataL(:);varDataR(:)];
                values = values(isfinite(values));
                if ~isempty(values)
                    limmin = min(values);
                    limmax = max(values);
                    if limmin == limmax
                        padding = max(1,abs(limmin)*0.1);
                    else
                        padding = (limmax-limmin)*0.1;
                    end
                    limits = [limmin-padding limmax+padding];
                    for sideIndex = 1:2
                        yyaxis(app.UIAxesAnalysis,sides{sideIndex});
                        app.UIAxesAnalysis.YLim = limits;
                    end
                end
            end
            index = round(app.TimesSlider.Value);
            setXlines(app,app.time(index));
            yyaxis(app.UIAxesAnalysis,pos);
            isSuccess = true;
            catch ME
                app.Lamp.Color = "red";
                setTexttarea(app,{"解析処理失敗",ME.identifier,ME.message});
            end
        end

        % 左右の解析設定を適用したデータを取得する
        % 引数 pos: "left" または "right"
        % 戻り値 varName: 選択列名、varData: 加工後データ（none時は空）
        function [varName,varData] = getAnalysisData(app,pos)
            if strcmp(pos,"left")
                varName = app.DataAnalysisLDropDown.Value;
                method = app.smoothLDropDown.Value;
                window = app.windowLSpinner.Value;
                normalizeState = app.normalizeLSwitch.Value;
                detrendState = app.detrendLSwitch.Value;
            else
                varName = app.DataAnalysisRDropDown.Value;
                method = app.smoothRDropDown.Value;
                window = app.windowRSpinner.Value;
                normalizeState = app.normalizeRSwitch.Value;
                detrendState = app.detrendRSwitch.Value;
            end
            varData = [];
            if strcmp(varName,"none")
                return;
            end
            varData = getNumericLogColumn(app,varName);
            if ~strcmp(method,"none")
                varData = smoothdata(varData,method,window);
            end
            if strcmp(normalizeState,"On")
                varData = normalize(varData);
            end
            if strcmp(detrendState,"On")
                varData = detrend(varData);
            end
            if any(~isfinite(varData))
                error("logAnalysis:InvalidAnalysis","列%sの加工結果にInfまたはNaNがあります。",varName);
            end
        end
        % ログ行のインデックスで現在地・スライダー・定数線を更新する
        % 引数 index: 選択行番号（整数化して有効範囲に制限）
        function selectCourseIndex(app,index)
            if isempty(app.time) || isempty(app.course) || ...
                    ~isnumeric(index) || ~isscalar(index) || ~isfinite(index)
                return;
            end
            rowCount = min(numel(app.time),height(app.course));
            lowerIndex = max(1,ceil(app.TimesSlider.Limits(1)));
            upperIndex = min(rowCount,floor(app.TimesSlider.Limits(2)));
            if lowerIndex > upperIndex
                return;
            end
            index = min(upperIndex,max(lowerIndex,round(index)));
            app.TimesSlider.Value = index;
            setXlines(app,app.time(index));
            updateLineSensorImage(app,index);

            % 時刻の重複に依存しない行番号で座標を選択する
            x = app.course.x(index);
            y = app.course.y(index);
            if ~isfinite(x) || ~isfinite(y)
                return;
            end
            if ~isempty(app.marker) && isgraphics(app.marker)
                app.marker.XData = x;
                app.marker.YData = y;
            else
                previousHold = ishold(app.UIAxisCourse);
                hold(app.UIAxisCourse,'on');
                app.marker = scatter(app.UIAxisCourse,x,y,"m",'filled', ...
                    'HitTest','off','PickableParts','none');
                if ~previousHold
                    hold(app.UIAxisCourse,'off');
                end
            end
        end

        % 選択行を中心に前後2行のラインセンサー値を白黒画像へ変換する
        % 引数 index: 選択中のログ行。0番が左端、4096が白、0が黒
        function updateLineSensorImage(app,index)
            if isempty(app.LineSensorAxes) || ~isgraphics(app.LineSensorAxes)
                return;
            end
            rowIndices = index + (2:-1:-2); % 上が後、中央が現在、下が前
            validRows = rowIndices >= 1 & rowIndices <= height(app.log);
            colors = repmat(0.75,5,10,3); % 行不足・不正値は灰色
            labels = repmat({'--'},1,5);
            for row = find(validRows)
                labels{row} = sprintf('%d (%.3fs)',rowIndices(row),app.time(rowIndices(row)));
            end
            names = cellstr("lSensorCari" + string(0:9));
            columnsPresent = all(ismember(names,app.log.Properties.VariableNames));
            columnsNumeric = columnsPresent;
            if columnsPresent
                columnsNumeric = all(cellfun(@(name)isnumeric(app.log.(name)) && ...
                    isreal(app.log.(name)) && iscolumn(app.log.(name)),names));
            end
            if columnsNumeric
                values = double(app.log{rowIndices(validRows),names});
                gray = min(4096,max(0,values)) / 4096;
                gray(~isfinite(values)) = 0.75;
                colors(validRows,:,:) = repmat(gray,1,1,3);
                app.LineSensorAxes.Title.String = 'Line sensors (0: black / 4096: white)';
            else
                app.LineSensorAxes.Title.String = 'Line sensor columns unavailable';
            end
            if isempty(app.lineSensorImage) || ~isgraphics(app.lineSensorImage)
                previousNextPlot = app.LineSensorAxes.NextPlot;

        % クリック元の軸と座標から最も近いログ行を選択する
        % 引数 src: クリックされた描画オブジェクト、event: クリックイベント
        function getTimeIndex(app,src,event)
            if isempty(app.time) || isempty(app.course) || ~isgraphics(src)
                return;
            end
            sourceAxes = ancestor(src,'axes');
            if isempty(sourceAxes)
                return;
            end
            clickPoint = sourceAxes.CurrentPoint;
            % クリックイベントの座標がある場合はその値を使用する
            if (isstruct(event) && isfield(event,'IntersectionPoint')) || ...
                    (isobject(event) && isprop(event,'IntersectionPoint'))
                point = event.IntersectionPoint;
                if isnumeric(point) && numel(point) >= 2 && all(isfinite(point(1:2)))
                    clickPoint(1,1:2) = point(1:2);
                end
            end
            X = src.XData(:);
            if isa(src,'matlab.graphics.chart.primitive.Scatter') && ...
                    isequal(sourceAxes,app.UIAxisCourse)
                Y = src.YData(:);
                distances = (X-clickPoint(1,1)).^2 + (Y-clickPoint(1,2)).^2;
            elseif isa(src,'matlab.graphics.chart.primitive.Line') && ...
                    (isequal(sourceAxes,app.UIAxes1) || ...
                     isequal(sourceAxes,app.UIAxes2) || ...
                     isequal(sourceAxes,app.UIAxes3) || ...
                     isequal(sourceAxes,app.UIAxes4) || ...
                     isequal(sourceAxes,app.UIAxes5))
                if ~strcmp(app.graphPositionBotton.Value,"On")
                    return;
                end
                distances = abs(X-clickPoint(1,1));
            else
                return;
            end
            distances(~isfinite(distances)) = Inf;
            if isempty(distances) || all(isinf(distances))
                return;
            end
            [~,index] = min(distances);
            selectCourseIndex(app,index);
        end

        % 引数 graphAxes: 描画先の軸、dataDropDown: 表示列の選択UI
        function isSuccess = plotVariableGraph(app,graphAxes,dataDropDown)
            isSuccess = false;
            if isempty(app.log) || isempty(app.time)
                return;
            end
            try
            data = getNumericLogColumn(app,dataDropDown.Value);
            axesList = {app.UIAxes1,app.UIAxes2,app.UIAxes3,app.UIAxes4,app.UIAxes5};
            plotIndex = find(cellfun(@(axesHandle)isequal(axesHandle,graphAxes),axesList),1);
            app.variablePlots{plotIndex} = updateDataLine(app,graphAxes, ...
                app.variablePlots{plotIndex},data);
            if strcmp(app.graphPositionBotton.Value,"On")
                app.variablePlots{plotIndex}.ButtonDownFcn = @(src,evnt)getTimeIndex(app,src,evnt);

                app.variablePlots{plotIndex}.ButtonDownFcn = [];
            end
            graphAxes.YLimMode = 'auto';
            title(graphAxes,dataDropDown.Value);
            graphAxes.Title.Visible = "off";
            isSuccess = true;
            catch ME
                app.Lamp.Color = "red";
                setTexttarea(app,{"グラフ描画失敗",ME.identifier,ME.message});
            end
        end

        function isSuccess = plotFlagPoint(app)
            isSuccess = false;
            if isempty(app.log) || isempty(app.course)
                return;
            end
            try
                colors = {'red','green','blue'};
                for itemIndex = 1:3
                    propertyName = sprintf('flag%d',itemIndex);
                    plotHandle = app.(propertyName);
                    enabled = app.(sprintf('Plotflag%denableCheckBox',itemIndex)).Value;
                    if ~enabled
                        if ~isempty(plotHandle) && isgraphics(plotHandle)
                            plotHandle.Visible = 'off';
                        end
                        continue;
                    end
                    columnName = app.(sprintf('Plotflag%dDropDown',itemIndex)).Value;
                    selected = getNumericLogColumn(app,columnName) > 0;
                    x = app.course.x(selected);
                    y = app.course.y(selected);

                        previousNextPlot = app.UIAxisCourse.NextPlot;
                        app.UIAxisCourse.NextPlot = 'add';

                        app.(propertyName) = scatter(app.UIAxisCourse,x,y,colors{itemIndex}, ...
                            'filled','HitTest','off','PickableParts','none');
                        clear cleanupAxes;
                    else
                        set(plotHandle,'XData',x,'YData',y,'Visible','on');
                    end
                end
                isSuccess = true;
            catch ME
                app.Lamp.Color = "red";
                setTexttarea(app,{"フラグ描画失敗",ME.identifier,ME.message});
            end

    % App creation and deletion
    methods (Access = public)

        % Construct app
        function app = logAnalysis(varargin)

            % Create UIFigure and components
            createComponents(app)

            % Register the app with App Designer
            registerApp(app, app.UIFigure)

            % Execute the startup function
            runStartupFcn(app, @(app)startupFcn(app, varargin{:}))

            if nargout == 0
                clear app
            end
        end

        % Code that executes before app deletion
        function delete(app)

            % Delete UIFigure when app is deleted
            delete(app.UIFigure)
        end
    end
