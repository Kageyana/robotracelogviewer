        % ログデータを取得する
        function isSuccess = getLogdata(app)
            isSuccess = false; % 全処理が成功するまで未完了とする

			try
				% ログファイル読み込みと最低限の行数チェックを例外処理で保護
                if isempty(app.LogDropDown.ItemsData)
                    error("logAnalysis:NoLogFiles","読み込み可能なログファイルがありません。");
                end
                filePath = fullfile(app.rootDir,app.LogDropDown.Value);
				logData = readtable(filePath);		  % ログファイル読み込み
                meta = readcell(filePath,"Range","A1:AAAA1");         % パラメータを取得
                colum = readcell(filePath,"Range","A2:AAAA2");         % 列名を取得 
                rowNum = size(logData.Properties.VariableNames,2);      % 有効なヘッダ数を取得
                logData.Properties.VariableNames = colum(1,1:rowNum);  %ヘッダを設定

                % パラメータも一時変数で保持する
                parameterData = meta;

                if height(logData) < 2
                    error("logAnalysis:InsufficientLogRows","ログファイルの行数が2行未満です。");
                end

                % テキストフィールドから換算式を取得
				% 時間行列[s]
				timeeq = split(app.timeEditField.Value);
				timeData = logData.(timeeq{1});
				i = 2;
				while i <= numel(timeeq)
					if i + 1 > numel(timeeq)
						error("logAnalysis:InvalidEquation","時間換算式の演算子と値の組み合わせが不足しています。");
					end
					operand = str2double(timeeq{i + 1});
					if isnan(operand)
						error("logAnalysis:InvalidEquation","時間換算式に数値以外が指定されています: %s", timeeq{i + 1});
					end
					% 演算子を判定して掛け算・割り算を実行
					if timeeq(i) == "*"
						timeData = timeData * operand;
					elseif timeeq(i) == "/"
						timeData = timeData / operand;
					else
						error("logAnalysis:InvalidEquation","時間換算式に無効な演算子があります: %s", timeeq{i});
					end
					i = i + 2;
                end
                dtData = diff(timeData);
                dtData = [dtData;dtData(end)];

				% 速度行列[mm/s] の換算式を検証しながら計算
				velocityeq = split(app.velocityEditField.Value);
				velocityData = logData.(velocityeq{1});
                if app.diffCheckBox.Value == 1
                    m = diff(velocityData);
                    n = [m;m(end)];
                    o = n ./ dtData;

                    velocityData = o;
                end
				i = 2;
				while i <= numel(velocityeq)
					if i + 1 > numel(velocityeq)
						error("logAnalysis:InvalidEquation","速度換算式の演算子と値の組み合わせが不足しています。");
					end
					velocityOperand = str2double(velocityeq{i + 1});
					if isnan(velocityOperand)
						error("logAnalysis:InvalidEquation","速度換算式に数値以外が指定されています: %s", velocityeq{i + 1});
					end
					% 速度換算式の演算子を判定して計算
					if velocityeq(i) == "*"
						velocityData = velocityData * velocityOperand;
					elseif velocityeq(i) == "/"
						velocityData = velocityData / velocityOperand;
					else
						error("logAnalysis:InvalidEquation","速度換算式に無効な演算子があります: %s", velocityeq{i});
					end
					i = i + 2;
				end

				% 角度速度配列[deg/s] の換算式を検証しながら計算（右旋回時にプラス）
				angularVelocityeq = split(app.angularVelocityEditField.Value);
				angularVelocityData = logData.(angularVelocityeq{1});
				i = 2;
                while i <= numel(angularVelocityeq)
                    if i + 1 > numel(angularVelocityeq)
                        error("logAnalysis:InvalidEquation","角速度換算式の演算子と値の組み合わせが不足しています。");
                    end
                    angularVelocityOperand = str2double(angularVelocityeq{i + 1});
                    if isnan(angularVelocityOperand)
                        error("logAnalysis:InvalidEquation","角速度換算式に数値以外が指定されています: %s", angularVelocityeq{i + 1});
                    end
                    % 角速度換算式の演算子を判定して計算
                    if angularVelocityeq(i) == "*"
                        angularVelocityData = angularVelocityData * angularVelocityOperand;
                    elseif angularVelocityeq(i) == "/"
                        angularVelocityData = angularVelocityData / angularVelocityOperand;
                    else
                        error("logAnalysis:InvalidEquation","角速度換算式に無効な演算子があります: %s", angularVelocityeq{i});
                    end
                    i = i + 2;
                end
                if isempty(parameterData)
                    app.TextArea_2.Value = "";
                else
                    app.TextArea_2.Value = parameterData;
                end
                % 全ての読み込み・換算が成功した後にまとめて反映する
                app.log = logData;
                app.time = timeData;
                app.dt = dtData;
                app.velocity = velocityData;
                app.angularVelocity = angularVelocityData;
                app.parameter = parameterData;
                isSuccess = true;

			catch ME
				app.Lamp.Color = "red";	% 例外発生をランプで通知
				setTexttarea(app, {ME.identifier, ME.message});	% エラー内容をテキストエリアに表示
				isSuccess = false;
			end

			% 例外時は後続処理をスキップ
			if ~isSuccess
				return;
			end

			setTexttarea(app, {filePath, 'ログファイル読み込み完了'});
        end
        

        % ログファイル一覧の取得
        function isSuccess = getLogfiles(app)
            isSuccess = false;
            selectedFile = string(app.LogDropDown.Value);
            try
                if ~isfolder(app.rootDir)
                    error("MATLAB:cd:NonExistentFolder","The folder does not exist.");
                end
                fileList = struct2table(dir(fullfile(app.rootDir,'*.csv')));
                if isempty(fileList)
                    error("logAnalysis:NoLogFiles","読み込み可能なログファイルがありません。");
                end
                % ソート用の番号と読み込み用の元ファイル名を分離する
                fileNames = string(fileList.name);
                logNumbers = str2double(erase(fileNames,lettersPattern | characterListPattern(".")));
                valid = ~isnan(logNumbers) & fileList.bytes ~= 0;
                fileList = fileList(valid,:);
                fileNames = fileNames(valid);
                logNumbers = logNumbers(valid);
                if isempty(fileNames)
                    error("logAnalysis:NoLogFiles","読み込み可能なログファイルがありません。");
                end
                [~,order] = sort(logNumbers,'descend');
                fileList = fileList(order,:);
                fileNames = fileNames(order);
                logNumbers = logNumbers(order);
                fileSize = string(round(fileList.bytes/1000));
                fileDate = string(datetime(fileList.datenum,"ConvertFrom","datenum"),"yyyy/MM/dd");

                % 表示文字列が変わってもファイル名で選択を保持する
                app.LogDropDown.ItemsData = {};
                app.LogDropDown.Items = append(string(logNumbers),"    ",fileSize,"KB","    ",fileDate);
                app.LogDropDown.ItemsData = cellstr(fileNames);
                if ismember(selectedFile,fileNames)
                    app.LogDropDown.Value = char(selectedFile);
                else
                    % 選択ログが削除された場合は一覧の先頭を選ぶ
                    app.LogDropDown.Value = char(fileNames(1));
                end
                app.LogDropDown.Enable = "on";
                app.Lamp.Color = "green";
                app.TextArea.Value = "";
                isSuccess = true;
            catch ME
                app.LogDropDown.ItemsData = {};
                app.LogDropDown.Items = {'not found files'};
                app.LogDropDown.Enable = "off";
                app.Lamp.Color = "red";
                setTexttarea(app,{ME.identifier,ME.message});
                if strcmp(ME.identifier,'MATLAB:cd:NonExistentFolder')
                    app.FolderButton.Text = "not found folder";
                end
            end
        end

        % コース座標の計算
        function isSuccess = calcCourseplot(app)
            isSuccess = false;
            if isempty(app.time) || isempty(app.velocity) || isempty(app.angularVelocity)
                return;
            end
            try
                % 換算式の出力単位に合わせて角度を積算する
                angle = cumtrapz(app.time,app.angularVelocity .* app.kgyro);
                if app.radsCheckBox.Value
                    angle = angle + deg2rad(app.rotate);
                    x = cumtrapz(app.time,app.velocity .* sin(angle));
                    y = cumtrapz(app.time,app.velocity .* cos(angle));
                else
                    angle = angle + app.rotate;
                    x = cumtrapz(app.time,app.velocity .* sind(angle));
                    y = cumtrapz(app.time,app.velocity .* cosd(angle));
                end
                % timetable変数作成
                app.course = timetable(seconds(app.time),x,y,app.velocity,'VariableNames',{'x','y','velocity'});
                app.shortCutCourse = []; % 基準軌跡が変わったら旧座標を無効にする
                isSuccess = true;
            catch ME
                app.Lamp.Color = "red";     % ステータスランプ赤
                % エラー表示
                for num = 1:1:length(ME.stack)
                    setTexttarea(app, {ME.stack(num,1).name,num2str(ME.stack(num,1).line)})
                end
                setTexttarea(app, {ME.identifier,ME.message})
            end
        end

        % ショートカット軌跡座標の計算
        function isSuccess = calcShortCutplot(app)
            isSuccess = false;
            app.shortCutCourse = [];
            if isempty(app.course)
                return;
            end
            try
                % windowが1未満の場合は処理を中断
                app.ShortCutwindowSlider.Value = round(app.ShortCutwindowSlider.Value);
                if app.ShortCutwindowSlider.Value < 1
                    app.Lamp.Color = "red";     % ステータスランプ赤
                    setTexttarea(app, {"windowの値が1未満です", "1以上に設定してください"});
                    return;
                end
                % 座標算出
                x = movmean(app.course.x,app.ShortCutwindowSlider.Value);
                y = movmean(app.course.y,app.ShortCutwindowSlider.Value);
                % timetable変数作成
                app.shortCutCourse = timetable(app.course.Time,x,y,app.course.velocity,'VariableNames',{'x','y','velocity'});
                isSuccess = true;
            catch ME
                app.Lamp.Color = "red";     % ステータスランプ赤
                % エラー表示
                for num = 1:1:length(ME.stack)
                    setTexttarea(app, {ME.stack(num,1).name,num2str(ME.stack(num,1).line)})
                end
                setTexttarea(app, {ME.identifier,ME.message})
            end
        end
       
        % コース描画
        function plotCourse(app)
            if isempty(app.course)
                return;
            end
            % 表示するたびに現在の基準軌跡とwindowから座標を更新する
            if strcmp(app.ShortCutplotSwitch.Value,"On") && ~calcShortCutplot(app)
                return;
            end
            try
                % 既存カラーバーがある場合は削除して重複表示を防止
                if ~isempty(app.courseColorbar) && isvalid(app.courseColorbar)
                    delete(app.courseColorbar);
                end
                % プロット
                scatter(app.UIAxisCourse,app.course,'x','y','ColorVariable','velocity','ButtonDownFcn',@(src,evnt)getTimeIndex(app,src,evnt));
                app.UIAxisCourse.XLim = [min(app.course.x)-100 max(app.course.x)+100];
                app.UIAxisCourse.YLim = [min(app.course.y)-100 max(app.course.y)+100];
                daspect(app.UIAxisCourse,[1 1 1]); % x,y軸の縦横比を1:1にする

                % コース軸の設定
                % カラーバー表示
                c = colorbar(app.UIAxisCourse);
                c.Label.String = 'velocity[mm/s]';
                c.Label.FontSize = 12;
                app.courseColorbar = c; % 新規作成したカラーバーのハンドルを保持

                % ショートカット軌跡表示ボタンがONの時
                if app.ShortCutplotSwitch.Value == "On"
                    hold(app.UIAxisCourse, 'on');
                    scatter(app.UIAxisCourse,app.shortCutCourse.x,app.shortCutCourse.y, ...
                        'HitTest','off','PickableParts','none');
                    hold(app.UIAxisCourse, 'off');
                end
                
                plotFlagPoint(app);
                selectCourseIndex(app,app.TimesSlider.Value);
            catch ME
                app.Lamp.Color = "red";     % ステータスランプ赤
                % エラー表示
                for num = 1:1:length(ME.stack)
                    % setTexttarea(app, {ME.stack(num,1).name,num2str(ME.stack(num,1).line)})
                end
                setTexttarea(app, {ME.identifier,ME.message});
            end
        end

        % ログの取得、再プロット
        function isSuccess = reload(app,EventName)
            isSuccess = false;
            if isempty(app.rootDir) || isequal(app.rootDir,0)
                app.rootDir = uigetdir;
            end
            if ~isempty(app.rootDir) && ~isequal(app.rootDir,0)
                if strcmp(EventName,"ButtonPushed")
                    if ~getLogfiles(app)
                        return;
                    end
                end
                if ~getLogdata(app)
                    return;
                end
                getLogvalues(app);  % ログ変数一覧を取得

                
    
                % プロット初期化
                cla(app.UIAxisCourse)
                cla(app.UIAxes1)
                cla(app.UIAxes2)
                cla(app.UIAxes3)
                cla(app.UIAxesAnalysis)
    
                % 変数プロットのx軸設定
                app.UIAxes1.XLim = [0 app.time(end,1)];
                app.UIAxes2.XLim = [0 app.time(end,1)];
                app.UIAxes3.XLim = [0 app.time(end,1)];
                app.UIAxesAnalysis.XLim = [0 app.time(end,1)];
    
                % スライダー範囲の設定
                app.TimesSlider.Limits = [1 numel(app.time)];   % 時間配列のインデックスをスライダーの範囲に設定
                Ticks = round(linspace(1,numel(app.time),17));  % インデックスを17等分する
                app.TimesSlider.MajorTicks = Ticks;             % 主目盛りを設定
                TicksLabelnum = app.time(Ticks');               % ラベルにする秒数を時間配列から取得
                TicksLabelstr = num2str(TicksLabelnum);         % 秒数を文字列に変換
                TicksLabel = cellstr(TicksLabelstr)';           % cell配列に変換
                app.TimesSlider.MajorTickLabels = TicksLabel;   % ラベルを設定
    
                % 変数初期化
                % app.kgyro = 1;
                % app.CourseajustgyroKnob.Value = 1;
                % app.Spinner.Value = 1;
    
                % コース座標計算
                calcCourseplot(app);    % コース座標計算
                
                % コース描画
                plotCourse(app)
    
                % 変数プロット
                plotVariableGraphs(app);

                % 変数解析プロット
                procData(app,"left");

                % 選択フラグのプロット
                plotFlagPoint(app);
                isSuccess = true;
            end
        end

        % データ処理
        function procData(app,pos)
            % 両側を現在の設定で処理し、描画データと範囲計算を一致させる
            [varNameL,varDataL] = getAnalysisData(app,"left");
            [varNameR,varDataR] = getAnalysisData(app,"right");
            sides = {"left","right"};
            names = {varNameL,varNameR};
            data = {varDataL,varDataR};
            for sideIndex = 1:2
                yyaxis(app.UIAxesAnalysis,sides{sideIndex});
                ylabel(app.UIAxesAnalysis,names{sideIndex});
                if strcmp(names{sideIndex},"none")
                    cla(app.UIAxesAnalysis);
                else
                    plot(app.UIAxesAnalysis,app.time,data{sideIndex});
                end
                app.UIAxesAnalysis.YLimMode = "auto";
            end

            if strcmp(app.ylimmatchSwitch.Value,"On") && ...
                    ~strcmp(varNameL,"none") && ~strcmp(varNameR,"none")
                values = [varDataL(:);varDataR(:)];
                values = values(isfinite(values));
                if ~isempty(values)
                    limmin = min(values);
                    limmax = max(values);
                    if limmin == limmax
                        padding = max(1,abs(limmin)*0.1);
                    else
                        padding = (limmax-limmin)*0.1;
                    end
                    limits = [limmin-padding limmax+padding];
                    for sideIndex = 1:2
                        yyaxis(app.UIAxesAnalysis,sides{sideIndex});
                        app.UIAxesAnalysis.YLim = limits;
                    end
                end
            end
            index = round(app.TimesSlider.Value);
            setXlines(app,app.time(index));
            yyaxis(app.UIAxesAnalysis,pos);
        end

        % 左右の解析設定を適用したデータを取得する
        % 引数 pos: "left" または "right"
        % 戻り値 varName: 選択列名、varData: 加工後データ（none時は空）
        function [varName,varData] = getAnalysisData(app,pos)
            if strcmp(pos,"left")
                varName = app.DataAnalysisLDropDown.Value;
                method = app.smoothLDropDown.Value;
                window = app.windowLSpinner.Value;
                normalizeState = app.normalizeLSwitch.Value;
                detrendState = app.detrendLSwitch.Value;
            else
                varName = app.DataAnalysisRDropDown.Value;
                method = app.smoothRDropDown.Value;
                window = app.windowRSpinner.Value;
                normalizeState = app.normalizeRSwitch.Value;
                detrendState = app.detrendRSwitch.Value;
            end
            varData = [];
            if strcmp(varName,"none")
                return;
            end
            varData = app.log.(varName);
            if ~strcmp(method,"none")
                varData = smoothdata(varData,method,window);
            end
            if strcmp(normalizeState,"On")
                varData = normalize(varData);
            end
            if strcmp(detrendState,"On")
                varData = detrend(varData);
            end
        end
        % ログ行のインデックスで現在地・スライダー・定数線を更新する
        % 引数 index: 選択行番号（整数化して有効範囲に制限）
        function selectCourseIndex(app,index)
            if isempty(app.time) || isempty(app.course) || ...
                    ~isnumeric(index) || ~isscalar(index) || ~isfinite(index)
                return;
            end
            rowCount = min(numel(app.time),height(app.course));
            lowerIndex = max(1,ceil(app.TimesSlider.Limits(1)));
            upperIndex = min(rowCount,floor(app.TimesSlider.Limits(2)));
            if lowerIndex > upperIndex
                return;
            end
            index = min(upperIndex,max(lowerIndex,round(index)));
            app.TimesSlider.Value = index;
            setXlines(app,app.time(index));

            % 時刻の重複に依存しない行番号で座標を選択する
            x = app.course.x(index);
            y = app.course.y(index);
            if ~isfinite(x) || ~isfinite(y)
                return;
            end
            if ~isempty(app.marker) && isgraphics(app.marker)
                app.marker.XData = x;
                app.marker.YData = y;
            else
                previousHold = ishold(app.UIAxisCourse);
                hold(app.UIAxisCourse,'on');
                app.marker = scatter(app.UIAxisCourse,x,y,"m",'filled', ...
                    'HitTest','off','PickableParts','none');
                if ~previousHold
                    hold(app.UIAxisCourse,'off');
                end
            end
        end

        % クリック元の軸と座標から最も近いログ行を選択する
        % 引数 src: クリックされた描画オブジェクト、event: クリックイベント
        function getTimeIndex(app,src,event)
            if isempty(app.time) || isempty(app.course) || ~isgraphics(src)
                return;
            end
            sourceAxes = ancestor(src,'axes');
            if isempty(sourceAxes)
                return;
            end
            clickPoint = sourceAxes.CurrentPoint;
            % クリックイベントの座標がある場合はその値を使用する
            if (isstruct(event) && isfield(event,'IntersectionPoint')) || ...
                    (isobject(event) && isprop(event,'IntersectionPoint'))
                point = event.IntersectionPoint;
                if isnumeric(point) && numel(point) >= 2 && all(isfinite(point(1:2)))
                    clickPoint(1,1:2) = point(1:2);
                end
            end
            X = src.XData(:);
            if isa(src,'matlab.graphics.chart.primitive.Scatter') && ...
                    isequal(sourceAxes,app.UIAxisCourse)
                Y = src.YData(:);
                distances = (X-clickPoint(1,1)).^2 + (Y-clickPoint(1,2)).^2;
            elseif isa(src,'matlab.graphics.chart.primitive.Line') && ...
                    (isequal(sourceAxes,app.UIAxes1) || ...
                     isequal(sourceAxes,app.UIAxes2) || ...
                     isequal(sourceAxes,app.UIAxes3))
                if ~strcmp(app.graphPositionBotton.Value,"On")
                    return;
                end
                distances = abs(X-clickPoint(1,1));
            else
                return;
            end
            distances(~isfinite(distances)) = Inf;
            if isempty(distances) || all(isinf(distances))
                return;
            end
            [~,index] = min(distances);
            selectCourseIndex(app,index);
        end

        % 変数グラフを描画し、位置選択設定に応じてクリック処理を登録する
        % 引数 graphAxes: 描画先の軸、dataDropDown: 表示列の選択UI
        function plotVariableGraph(app,graphAxes,dataDropDown)
            if isempty(app.log) || isempty(app.time)
                return;
            end
            if strcmp(app.graphPositionBotton.Value,"On")
                plot(graphAxes,app.time,app.log.(dataDropDown.Value), ...
                    'ButtonDownFcn',@(src,evnt)getTimeIndex(app,src,evnt));
            else
                plot(graphAxes,app.time,app.log.(dataDropDown.Value));
            end
            title(graphAxes,dataDropDown.Value);
            graphAxes.Title.Visible = "off";
        end

            linkaxes([app.UIAxes1 app.UIAxes2 app.UIAxes3],'x');
            selectCourseIndex(app,app.TimesSlider.Value);
        end

        % Code that executes after component creation
        function startupFcn(app, arg1)
            % 起動時
            try
                % 設定ファイルを選択する
                F = msgbox("設定ファイルを選択してください");
                [filePath,location] = uigetfile({'*.csv'});
                close(F);
                % 設定用変数ファイルの読み込み（CSV）
                if ~isequal(filePath,0)
                    filePath = append(location,filePath);
                    settingsTable = readtable(filePath,"TextType","string");   % Table型に変換
                    settingsMap = containers.Map(settingsTable.key, settingsTable.value);   % キーマップ作成
                    
                    % 換算式をCSVから復元
                    app.timeEditField.Value = settingsMap("timeEquation");
                    app.velocityEditField.Value = settingsMap("velocityEquation");
                    app.angularVelocityEditField.Value = settingsMap("angularVelocityEquation");
                    app.radsCheckBox.Value = lower(settingsMap("radsCheckBox")) == "true";
                    
                    % 解析・ショートカット設定をCSVから復元
                    app.SwitchDirection.Value = settingsMap("switchDirection");
                    app.ShortCutplotSwitch.Value = settingsMap("plotSwitch");
                    app.ylimmatchSwitch.Value = settingsMap("ylimMatch");
                    app.normalizeLSwitch.Value = settingsMap("normalizeLeft");
                    app.normalizeRSwitch.Value = settingsMap("normalizeRight");
                    app.detrendLSwitch.Value = settingsMap("detrendLeft");
                    app.detrendRSwitch.Value = settingsMap("detrendRight");
                    
                    % スピナー／スライダーの数値項目を文字列から復元
                    windowLeft = str2double(settingsMap("windowLeft"));
                    if ~isnan(windowLeft)
                        app.windowLSpinner.Value = windowLeft;
                    end
                    windowRight = str2double(settingsMap("windowRight"));
                    if ~isnan(windowRight)
                        app.windowRSpinner.Value = windowRight;
                    end
                    sliderWindow = str2double(settingsMap("windowSlider"));
                    if ~isnan(sliderWindow)
                        app.ShortCutwindowSlider.Value = sliderWindow;
                    end
                    
                    % ログフォルダパスをCSVから復元
                    app.rootDir = settingsMap("logFolderDir");
                    str = strsplit(app.rootDir, filesep);
                    app.FolderButton.Text = str{end};

                    % 設定ファイルのパスをCSVから復元
                    app.csvFilePathEditer.Value = location;
                else
                    % 設定ファイルが無いときはログフォルダを選択する
                    F = msgbox("ログファイルの保存フォルダを選択してください");
                    app.rootDir = uigetdir;
                    close(F);
                end
            catch ME
                if (strcmp(ME.identifier,'MATLAB:load:couldNotReadFile'))
                    % 設定用変数ファイルが無い
                end
            end

            if ~isempty(app.rootDir) && ~isequal(app.rootDir,0)
		        if ~getLogfiles(app)
                    return;
                end
                if ~getLogdata(app)
                    return;
                end
		        getLogvalues(app);  % ログ変数一覧を取得
    
		        % 前回の表示グラフ名を取得
		        if ~isequal(filePath,0)
			        app.Data1DropDown.Value = settingsMap("data1");
			        app.Data2DropDown.Value = settingsMap("data2");
			        app.Data3DropDown.Value = settingsMap("data3");
                    
			        if isKey(settingsMap,"analysisLeft") && ismember(settingsMap("analysisLeft"), string(app.DataAnalysisLDropDown.Items))
				        app.DataAnalysisLDropDown.Value = settingsMap("analysisLeft");
			        end
			        if isKey(settingsMap,"analysisRight") && ismember(settingsMap("analysisRight"), string(app.DataAnalysisRDropDown.Items))
				        app.DataAnalysisRDropDown.Value = settingsMap("analysisRight");
			        end
			        if isKey(settingsMap,"smoothLeft")
				        app.smoothLDropDown.Value = settingsMap("smoothLeft");
			        end
			        if isKey(settingsMap,"smoothRight")
				        app.smoothRDropDown.Value = settingsMap("smoothRight");
			        end
		        end
        
		        % スライダー範囲の設定
                app.TimesSlider.Limits = [1 numel(app.time)];   % 時間配列のインデックスをスライダーの範囲に設定
                Ticks = round(linspace(1,numel(app.time),17));  % インデックスを17等分する
                app.TimesSlider.MajorTicks = Ticks;             % 主目盛りを設定
                TicksLabelnum = app.time(Ticks');               % ラベルにする秒数を時間配列から取得
                TicksLabelstr = num2str(TicksLabelnum);         % 秒数を文字列に変換
                TicksLabel = cellstr(TicksLabelstr)';           % cell配列に変換
                app.TimesSlider.MajorTickLabels = TicksLabel;   % ラベルを設定

		        % 変数プロットのx軸設定
		        app.UIAxes1.XLim = [0 app.time(end,1)];
		        app.UIAxes2.XLim = [0 app.time(end,1)];
		        app.UIAxes3.XLim = [0 app.time(end,1)];
		        app.UIAxesAnalysis.XLim = [0 app.time(end,1)];
        
		        % コース軸の設定
		        % カラーバー表示
		        if ~isempty(app.courseColorbar) && isvalid(app.courseColorbar)
			        delete(app.courseColorbar); % 既存カラーバーを削除して重複を回避
		        end
		        c = colorbar(app.UIAxisCourse);

		        c.Label.FontSize = 12;
		        app.courseColorbar = c; % 新規作成したカラーバーのハンドルを保持

			        app.UIAxisCourse.YLim = [app.courseSize(2,1) app.courseSize(2,2)];

			        app.UIAxisCourse.YLim = [app.courseSize(2,1) app.courseSize(2,2)];
		        end
		        app.UIAxisCourse.XTick = -10000:100:10000;
		        app.UIAxisCourse.YTick = -10000:100:10000;
        
		        % コース描画、変数グラフ表示
		        % コース座標計算
		        calcCourseplot(app);    % コース座標計算
        
		        % コース描画
		        plotCourse(app)
        
		        % 変数プロット
		        plotVariableGraphs(app);

                % 変数解析プロット
                procData(app,"left");

                if app.radsCheckBox.Value == 0
                    app.angularVelocitydegsLabel.Text = 'angularVelocity [deg/s] = ';

        function app = logAnalysis(varargin)

            % Create UIFigure and components

            % Register the app with App Designer
            registerApp(app, app.UIFigure)

            % Execute the startup function
            runStartupFcn(app, @(app)startupFcn(app, varargin{:}))

            if nargout == 0
                clear app
            end
        end

            % Delete UIFigure when app is deleted

        end
